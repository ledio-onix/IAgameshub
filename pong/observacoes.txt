const canvas = document.getElementById('pongCanvas');
const ctx = canvas.getContext('2d');

const scoreLeftEl = document.getElementById('score-left');
const scoreRightEl = document.getElementById('score-right');
const startBtn = document.getElementById('startBtn');
const resetBtn = document.getElementById('resetBtn');

const game = {
  running: false,
  speed: 7,
  paddleWidth: 14,
  paddleHeight: 90,
  leftY: (canvas.height - 90) / 2,
  rightY: (canvas.height - 90) / 2,
  leftScore: 0,
  rightScore: 0,
  ball: {
    x: canvas.width / 2,
    y: canvas.height / 2,
    radius: 10,
    speedX: 5,
    speedY: 4,
  },
};

const keys = {
  w: false,
  s: false,
  ArrowUp: false,
  ArrowDown: false,
};

function resetBall() {
  game.ball.x = canvas.width / 2;
  game.ball.y = canvas.height / 2;

  const direction = Math.random() < 0.5 ? -1 : 1;
  const angle = (Math.random() - 0.5) * 2.5;

  game.ball.speedX = direction * (5 + Math.random() * 1.5);
  game.ball.speedY = angle * 3;
}

function updateScoreboard() {
  scoreLeftEl.textContent = String(game.leftScore);
  scoreRightEl.textContent = String(game.rightScore);
}

function resetGame() {
  game.running = false;
  startBtn.textContent = 'Iniciar';
  game.leftScore = 0;
  game.rightScore = 0;
  game.leftY = (canvas.height - game.paddleHeight) / 2;
  game.rightY = (canvas.height - game.paddleHeight) / 2;
  updateScoreboard();
  resetBall();
}

function clampPaddle(value) {
  return Math.max(0, Math.min(canvas.height - game.paddleHeight, value));
}

function checkPaddleCollision() {
  const leftX = 20;
  const rightX = canvas.width - 20 - game.paddleWidth;
  const ball = game.ball;

  const hitsLeft =
    ball.x - ball.radius <= leftX + game.paddleWidth &&
    ball.x - ball.radius >= leftX &&
    ball.y >= game.leftY &&
    ball.y <= game.leftY + game.paddleHeight;

  const hitsRight =
    ball.x + ball.radius >= rightX &&
    ball.x + ball.radius <= rightX + game.paddleWidth &&
    ball.y >= game.rightY &&
    ball.y <= game.rightY + game.paddleHeight;

  if (hitsLeft || hitsRight) {
    const paddleY = hitsLeft ? game.leftY : game.rightY;
    const impact = (ball.y - (paddleY + game.paddleHeight / 2)) / (game.paddleHeight / 2);

    ball.speedX *= -1.08;
    ball.speedY = impact * 6;
  }
}

function updateGame() {
  if (keys.w) game.leftY -= game.speed;
  if (keys.s) game.leftY += game.speed;
  if (keys.ArrowUp) game.rightY -= game.speed;
  if (keys.ArrowDown) game.rightY += game.speed;

  game.leftY = clampPaddle(game.leftY);
  game.rightY = clampPaddle(game.rightY);

  if (!game.running) return;

  const ball = game.ball;
  ball.x += ball.speedX;
  ball.y += ball.speedY;

  if (ball.y - ball.radius <= 0 || ball.y + ball.radius >= canvas.height) {
    ball.speedY *= -1;
  }

  checkPaddleCollision();

  if (ball.x - ball.radius <= 0) {
    game.rightScore += 1;
    updateScoreboard();
    resetBall();
  }

  if (ball.x + ball.radius >= canvas.width) {
    game.leftScore += 1;
    updateScoreboard();
    resetBall();
  }
}

function drawNet() {
  ctx.strokeStyle = 'rgba(226, 232, 240, 0.7)';
  ctx.lineWidth = 3;
  ctx.setLineDash([12, 14]);
  ctx.beginPath();
  ctx.moveTo(canvas.width / 2, 0);
  ctx.lineTo(canvas.width / 2, canvas.height);
  ctx.stroke();
  ctx.setLineDash([]);
}

function drawPaddles() {
  ctx.fillStyle = '#38bdf8';
  ctx.fillRect(20, game.leftY, game.paddleWidth, game.paddleHeight);

  ctx.fillStyle = '#facc15';
  ctx.fillRect(canvas.width - 20 - game.paddleWidth, game.rightY, game.paddleWidth, game.paddleHeight);
}

function drawBall() {
  ctx.beginPath();
  ctx.arc(game.ball.x, game.ball.y, game.ball.radius, 0, Math.PI * 2);
  ctx.fillStyle = '#f8fafc';
  ctx.fill();
}

function draw() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  ctx.fillStyle = '#020817';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  drawNet();
  drawPaddles();
  drawBall();
}

function gameLoop() {
  updateGame();
  draw();
  requestAnimationFrame(gameLoop);
}

window.addEventListener('keydown', (event) => {
  if (event.key in keys) keys[event.key] = true;
});

window.addEventListener('keyup', (event) => {
  if (event.key in keys) keys[event.key] = false;
});

startBtn.addEventListener('click', () => {
  game.running = !game.running;
  startBtn.textContent = game.running ? 'Pausar' : 'Iniciar';
});

resetBtn.addEventListener('click', () => {
  resetGame();
});

resetGame();
gameLoop();
